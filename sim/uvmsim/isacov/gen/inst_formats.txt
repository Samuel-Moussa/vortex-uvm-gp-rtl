# Instruction OPERAND FORMAT -> the <ext>_sample() dispatch body.
#
# The dvplan CSV says WHICH coverpoints an instruction has, but not the ORDER
# its operands appear in the disassembly, nor whether an immediate is an address
# (add_imm_addr), a register offset (add_reg_offset), or a plain immediate; nor
# whether the instruction is a load/store that needs add_mem_address().  That is
# instruction-FORMAT knowledge from the ISA specification, supplied here as DATA.
#
# Syntax:  <mnemonic> = <op>[,<op>...] [| load|store]
#   ops: rd rs1 rs2 imm imm_addr reg_offset none
# Position in the list is the operand index passed to add_*().

# --- RV32I: R-type (rd, rs1, rs2) ---
add    = rd,rs1,rs2
sub    = rd,rs1,rs2
sll    = rd,rs1,rs2
slt    = rd,rs1,rs2
sltu   = rd,rs1,rs2
xor    = rd,rs1,rs2
srl    = rd,rs1,rs2
sra    = rd,rs1,rs2
or     = rd,rs1,rs2
and    = rd,rs1,rs2

# --- RV32I: I-type arithmetic (rd, rs1, imm) ---
addi   = rd,rs1,imm
slti   = rd,rs1,imm
sltiu  = rd,rs1,imm
xori   = rd,rs1,imm
ori    = rd,rs1,imm
andi   = rd,rs1,imm
slli   = rd,rs1,imm
srli   = rd,rs1,imm
srai   = rd,rs1,imm

# --- RV32I: loads  rd, imm(rs1) ---
lb     = rd,imm,rs1 | load
lh     = rd,imm,rs1 | load
lw     = rd,imm,rs1 | load
lbu    = rd,imm,rs1 | load
lhu    = rd,imm,rs1 | load

# --- RV32I: stores  rs2, imm(rs1) ---
sb     = rs2,imm,rs1 | store
sh     = rs2,imm,rs1 | store
sw     = rs2,imm,rs1 | store

# --- RV32I: branches (rs1, rs2, target) ---
beq    = rs1,rs2,imm_addr
bne    = rs1,rs2,imm_addr
blt    = rs1,rs2,imm_addr
bge    = rs1,rs2,imm_addr
bltu   = rs1,rs2,imm_addr
bgeu   = rs1,rs2,imm_addr

# --- RV32I: upper-immediate and jumps ---
lui    = rd,imm
auipc  = rd,imm
jal    = rd,imm_addr
jalr   = rd,reg_offset

# --- RV32I: no operands ---
fence  = none
nop    = none

# --- RV32M: all eight are R-type ---
mul    = rd,rs1,rs2
mulh   = rd,rs1,rs2
mulhsu = rd,rs1,rs2
mulhu  = rd,rs1,rs2
div    = rd,rs1,rs2
divu   = rd,rs1,rs2
rem    = rd,rs1,rs2
remu   = rd,rs1,rs2

# --- RV32Zicsr: rd, csr, rs1|imm ---
csrrw   = rd,csr,rs1
csrrs   = rd,csr,rs1
csrrc   = rd,csr,rs1
csrrwi  = rd,csr,imm
csrrsi  = rd,csr,imm
csrrci  = rd,csr,imm

# --- RV32Zifencei ---
fence.i = none

# --- RV32F: operand order from real objdump -M numeric,no-aliases output.
# The rounding-mode operand is printed ONLY when it is not `dyn`, e.g.
#   fadd.s f9,f15,f14           <- rm omitted => encoded rm = dyn (0b111)
#   fcvt.w.s x15,f15,rtz        <- rm explicit
# ops[] is a fixed ops_t[6] (RISCV_instruction_base.svh:33), so reading the
# absent slot is safe and yields "", which get_frm() maps to `dyn` -- exactly
# what the encoding means. So fp_rm is listed at its position unconditionally.
fadd.s    = fd,fs1,fs2,fp_rm
fsub.s    = fd,fs1,fs2,fp_rm
fmul.s    = fd,fs1,fs2,fp_rm
fdiv.s    = fd,fs1,fs2,fp_rm
fsqrt.s   = fd,fs1,fp_rm
fmadd.s   = fd,fs1,fs2,fs3,fp_rm
fmsub.s   = fd,fs1,fs2,fs3,fp_rm
fnmadd.s  = fd,fs1,fs2,fs3,fp_rm
fnmsub.s  = fd,fs1,fs2,fs3,fp_rm
fmin.s    = fd,fs1,fs2
fmax.s    = fd,fs1,fs2
fsgnj.s   = fd,fs1,fs2
fsgnjn.s  = fd,fs1,fs2
fsgnjx.s  = fd,fs1,fs2
feq.s     = rd,fs1,fs2
flt.s     = rd,fs1,fs2
fle.s     = rd,fs1,fs2
fclass.s  = rd,fs1
fcvt.w.s  = rd,fs1,fp_rm
fcvt.wu.s = rd,fs1,fp_rm
fcvt.s.w  = fd,rs1,fp_rm
fcvt.s.wu = fd,rs1,fp_rm
fmv.w.x   = fd,rs1
fmv.x.w   = rd,fs1
flw       = fd,imm,rs1 | load
fsw       = fs2,imm,rs1 | store

# --- RV32D: same structure as RV32F, doubles. Operand order from the RISC-V
# spec (objdump -M numeric,no-aliases renders D exactly like F).
# NOTE: fcvt.d.l/.lu, fcvt.l/.lu.d, fmv.d.x, fmv.x.d are RV64-ONLY -- they
# cannot be encoded or executed at XLEN=32. They are in Imperas' RV32D dvplan
# regardless; their covergroups are generated but are structurally unreachable
# here (waiver W-14).
fadd.d    = fd,fs1,fs2,fp_rm
fsub.d    = fd,fs1,fs2,fp_rm
fmul.d    = fd,fs1,fs2,fp_rm
fdiv.d    = fd,fs1,fs2,fp_rm
fsqrt.d   = fd,fs1,fp_rm
fmadd.d   = fd,fs1,fs2,fs3,fp_rm
fmsub.d   = fd,fs1,fs2,fs3,fp_rm
fnmadd.d  = fd,fs1,fs2,fs3,fp_rm
fnmsub.d  = fd,fs1,fs2,fs3,fp_rm
fmin.d    = fd,fs1,fs2
fmax.d    = fd,fs1,fs2
fsgnj.d   = fd,fs1,fs2
fsgnjn.d  = fd,fs1,fs2
fsgnjx.d  = fd,fs1,fs2
feq.d     = rd,fs1,fs2
flt.d     = rd,fs1,fs2
fle.d     = rd,fs1,fs2
fclass.d  = rd,fs1
fcvt.w.d  = rd,fs1,fp_rm
fcvt.wu.d = rd,fs1,fp_rm
fcvt.d.w  = fd,rs1,fp_rm
fcvt.d.wu = fd,rs1,fp_rm
fcvt.s.d  = fd,fs1,fp_rm
fcvt.d.s  = fd,fs1,fp_rm
fcvt.l.d  = rd,fs1,fp_rm
fcvt.lu.d = rd,fs1,fp_rm
fcvt.d.l  = fd,rs1,fp_rm
fcvt.d.lu = fd,rs1,fp_rm
fmv.d.x   = fd,rs1
fmv.x.d   = rd,fs1
fld       = fd,imm,rs1 | load
fsd       = fs2,imm,rs1 | store
